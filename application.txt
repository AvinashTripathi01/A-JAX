# app.py


import streamlit as st
import pandas as pd
from datetime import date
import os

from utils import *
from charts import *
from exercise import *

# =====================================================
# PAGE CONFIG
# =====================================================

st.set_page_config(
    page_title="Fitness Analytics App",
    layout="wide"
)

st.title("🏋️ Fitness Analytics App")

# =====================================================
# FILE PATHS
# =====================================================

WORKOUT_FILE = "data/Workout_Log.csv"
BODY_FILE = "data/Body_Log.csv"

os.makedirs("data", exist_ok=True)

# =====================================================
# LOAD EXERCISE ENGINE
# =====================================================

exercise_df = load_exercises()

# =====================================================
# WORKOUT INPUT SECTION
# =====================================================

st.header("🏋️ Log Workout")

# -----------------------------------------------------
# DATE
# -----------------------------------------------------

workout_date = st.date_input(
    "Workout Date",
    date.today()
)

# =====================================================
# EXERCISE SELECTOR
# =====================================================

exercise_names = sorted(

    exercise_df[
        "exercise_name"
    ]
    .dropna()
    .unique()
    .tolist()
)

selected_exercise = st.selectbox(

    "Select Exercise",

    exercise_names
)

# =====================================================
# EXERCISE DETAILS
# =====================================================

exercise_details = get_exercise_details(

    exercise_df,

    selected_exercise
)

# =====================================================
# AUTO FILLED VALUES
# =====================================================

muscle_group = exercise_details[
    "internal_muscle_group"
]

region = exercise_details[
    "internal_region"
]

equipment = exercise_details[
    "equipment"
]

mechanic = exercise_details[
    "mechanic"
]

level = exercise_details[
    "level"
]

primary_muscles = exercise_details[
    "primary_muscles"
]

secondary_muscles = exercise_details[
    "secondary_muscles"
]

images = exercise_details[
    "images"
]

instructions = exercise_details[
    "instructions"
]

# =====================================================
# EXERCISE INFORMATION
# =====================================================

col1, col2, col3 = st.columns(3)

with col1:

    st.info(f"Muscle Group: {muscle_group}")

    st.info(f"Region: {region}")

with col2:

    st.info(f"Equipment: {equipment}")

    st.info(f"Mechanic: {mechanic}")

with col3:

    st.info(f"Level: {level}")

# =====================================================
# MUSCLE DISPLAY
# =====================================================

st.write("### Primary Muscles")

st.write(primary_muscles)

st.write("### Secondary Muscles")

st.write(secondary_muscles)

# =====================================================
# IMAGE PREVIEW
# =====================================================

if len(images) > 0:

    image_path = f"data/exercise_data/{images[0]}"

    if os.path.exists(image_path):

        st.image(
            image_path,
            width=300
        )

# =====================================================
# INSTRUCTIONS
# =====================================================

with st.expander("Exercise Instructions"):

    for step in instructions:

        st.write(f"• {step}")

# =====================================================
# WORKOUT FORM
# =====================================================

with st.form("workout_form"):

    sets = st.number_input(
        "Sets",
        min_value=1,
        value=3
    )

    reps = st.number_input(
        "Reps",
        min_value=1,
        value=10
    )

    weight = st.number_input(
        "Weight (kg)",
        min_value=0.0,
        value=0.0
    )

    submitted = st.form_submit_button(
        "Save Workout"
    )

    # =================================================
    # SAVE WORKOUT
    # =================================================

    if submitted:

        new_workout = pd.DataFrame([{

            "Date": str(workout_date),

            "Exercise": selected_exercise,

            "Muscle Group": muscle_group,

            "Region": region,

            "Primary Muscles": ", ".join(primary_muscles),

            "Secondary Muscles": ", ".join(secondary_muscles),

            "Equipment": equipment,

            "Mechanic": mechanic,

            "Level": level,

            "Sets": sets,

            "Reps": reps,

            "Weight (kg)": weight
        }])

        # -------------------------------------------------
        # EXISTING FILE
        # -------------------------------------------------

        if os.path.exists(WORKOUT_FILE):

            old_workout = pd.read_csv(WORKOUT_FILE)

            updated_workout = pd.concat(
                [old_workout, new_workout],
                ignore_index=True
            )

        else:

            updated_workout = new_workout

        # -------------------------------------------------
        # SAVE CSV
        # -------------------------------------------------

        updated_workout.to_csv(
            WORKOUT_FILE,
            index=False
        )

        st.success("✅ Workout Saved!")

# =====================================================
# BODY INPUT SECTION
# =====================================================

st.header("📊 Log Body Data")

with st.form("body_form"):

    body_date = st.date_input(
        "Body Log Date",
        date.today()
    )

    body_weight = st.number_input(
        "Body Weight (kg)",
        min_value=0.0,
        value=0.0,
        step=0.1
    )

    calories = st.number_input(
        "Calories Intake (+/-)",
        value=0
    )

    protein = st.number_input(
        "Protein (g) (+/-)",
        value=0
    )

    calories_burned = st.number_input(
        "Calories Burned (+/-)",
        value=0
    )

    submitted_body = st.form_submit_button(
        "Save Body Data"
    )

    # =================================================
    # SAVE BODY DATA
    # =================================================

    if submitted_body:

        if os.path.exists(BODY_FILE):

            body_df = pd.read_csv(BODY_FILE)

        else:

            body_df = pd.DataFrame(columns=[

                "Date",
                "Weight",
                "Weight Entries",
                "Calories Intake",
                "Protein",
                "Calories Burned"
            ])

        # -------------------------------------------------
        # EXISTING DATE
        # -------------------------------------------------

        if str(body_date) in body_df["Date"].astype(str).values:

            idx = body_df[
                body_df["Date"].astype(str) == str(body_date)
            ].index[0]

            old_weight = float(
                body_df.loc[idx, "Weight"]
            )

            old_entries = int(
                body_df.loc[idx, "Weight Entries"]
            )

            new_avg_weight = (
                (old_weight * old_entries)
                + body_weight
            ) / (old_entries + 1)

            body_df.loc[idx, "Weight"] = round(
                new_avg_weight,
                2
            )

            body_df.loc[idx, "Weight Entries"] = (
                old_entries + 1
            )

            body_df.loc[idx, "Calories Intake"] += calories

            body_df.loc[idx, "Protein"] += protein

            body_df.loc[idx, "Calories Burned"] += calories_burned

        # -------------------------------------------------
        # NEW ENTRY
        # -------------------------------------------------

        else:

            new_body_row = pd.DataFrame([{

                "Date": str(body_date),

                "Weight": body_weight,

                "Weight Entries": 1,

                "Calories Intake": calories,

                "Protein": protein,

                "Calories Burned": calories_burned
            }])

            body_df = pd.concat(
                [body_df, new_body_row],
                ignore_index=True
            )

        # -------------------------------------------------
        # SAVE CSV
        # -------------------------------------------------

        body_df.to_csv(
            BODY_FILE,
            index=False
        )

        st.success("✅ Body Data Updated!")

# =====================================================
# LOAD DATA
# =====================================================

if os.path.exists(WORKOUT_FILE):

    workout_df = pd.read_csv(WORKOUT_FILE)

else:

    workout_df = pd.DataFrame()

if os.path.exists(BODY_FILE):

    body_df = pd.read_csv(BODY_FILE)

else:

    body_df = pd.DataFrame()

# =====================================================
# DATE FILTER
# =====================================================

if not workout_df.empty or not body_df.empty:

    st.divider()

    st.header("📅 Date Filter")

    all_dates = []

    if not workout_df.empty:

        workout_df["Date"] = pd.to_datetime(
            workout_df["Date"]
        )

        all_dates.extend(
            workout_df["Date"].tolist()
        )

    if not body_df.empty:

        body_df["Date"] = pd.to_datetime(
            body_df["Date"]
        )

        all_dates.extend(
            body_df["Date"].tolist()
        )

    min_date = min(all_dates)

    max_date = max(all_dates)

    start_date, end_date = st.date_input(

        "Select Date Range",

        value=(min_date, max_date)
    )

    # -------------------------------------------------
    # FILTER WORKOUT DATA
    # -------------------------------------------------

    if not workout_df.empty:

        workout_df = workout_df[

            (workout_df["Date"] >= pd.to_datetime(start_date))
            &
            (workout_df["Date"] <= pd.to_datetime(end_date))
        ]

    # -------------------------------------------------
    # FILTER BODY DATA
    # -------------------------------------------------

    if not body_df.empty:

        body_df = body_df[

            (body_df["Date"] >= pd.to_datetime(start_date))
            &
            (body_df["Date"] <= pd.to_datetime(end_date))
        ]

# =====================================================
# ANALYTICS PROCESSING
# =====================================================

if not workout_df.empty:

    workout_df = calculate_volume(workout_df)

    daily_df = daily_volume(workout_df)

    muscle_df = muscle_volume(workout_df)

else:

    daily_df = pd.DataFrame()

    muscle_df = pd.DataFrame()

if not body_df.empty:

    body_df = protein_adequacy(body_df)

    body_df = calorie_balance(body_df)

if not workout_df.empty and not body_df.empty:

    strength_df = strength_index(
        workout_df,
        body_df
    )

    growth_df = growth_score(
        body_df,
        strength_df
    )

else:

    strength_df = pd.DataFrame()

    growth_df = pd.DataFrame()

# =====================================================
# DASHBOARD KPI CARDS
# =====================================================

st.divider()

st.header("📊 Dashboard KPIs")

if not growth_df.empty:

    latest_growth = growth_df.iloc[-1]

    latest_strength = strength_df.iloc[-1]

    latest_daily = daily_df.iloc[-1]

    col1, col2, col3, col4, col5 = st.columns(5)

    with col1:

        st.metric(
            "Daily Volume",
            round(latest_daily["Volume"], 2)
        )

    with col2:

        st.metric(
            "Protein/kg",
            round(
                latest_growth["Protein Adequacy"],
                2
            )
        )

    with col3:

        st.metric(
            "Calorie Balance",
            round(
                latest_growth["Calorie Balance"],
                2
            )
        )

    with col4:

        st.metric(
            "Strength Index",
            round(
                latest_strength["Strength Index"],
                2
            )
        )

    with col5:

        st.metric(
            "Growth Score",
            round(
                latest_growth["Growth Score"],
                2
            )
        )

# =====================================================
# PERFORMANCE ANALYTICS
# =====================================================

st.divider()

st.header("🏋️ Performance Analytics")

col1, col2 = st.columns(2)

with col1:

    if not daily_df.empty:

        st.subheader("Daily Volume Trend")

        fig = plot_volume_trend(daily_df)

        st.pyplot(fig)

with col2:

    if not strength_df.empty:

        st.subheader("Strength Trend")

        fig = plot_strength_trend(strength_df)

        st.pyplot(fig)

if not daily_df.empty:

    st.subheader("Weekly Workload")

    fig = plot_weekly_workload(daily_df)

    st.pyplot(fig)

# =====================================================
# MUSCLE ANALYTICS
# =====================================================

st.divider()

st.header("💪 Muscle Analytics")

col3, col4 = st.columns(2)

with col3:

    if not muscle_df.empty:

        st.subheader("Muscle Distribution")

        fig = plot_muscle_distribution(muscle_df)

        st.pyplot(fig)

with col4:

    if not muscle_df.empty:

        st.subheader("Muscle Imbalance")

        fig = plot_muscle_imbalance(muscle_df)

        st.pyplot(fig)

# =====================================================
# BODY COMPOSITION ANALYTICS
# =====================================================

st.divider()

st.header("📉 Body Composition Analytics")

col5, col6 = st.columns(2)

with col5:

    if not body_df.empty:

        st.subheader("Weight Trend")

        fig = plot_weight_trend(body_df)

        st.pyplot(fig)

with col6:

    if not body_df.empty:

        st.subheader("Calories Trend")

        fig = plot_calorie_trend(body_df)

        st.pyplot(fig)

col7, col8 = st.columns(2)

with col7:

    if not body_df.empty:

        st.subheader("Protein Trend")

        fig = plot_protein_trend(body_df)

        st.pyplot(fig)

with col8:

    if not growth_df.empty:

        st.subheader("Growth Score Trend")

        fig = plot_growth_score(growth_df)

        st.pyplot(fig)

# =====================================================
# RAW TABLES
# =====================================================

st.divider()

with st.expander("📁 Workout Log"):

    if not workout_df.empty:

        st.dataframe(
            workout_df,
            use_container_width=True
        )

with st.expander("📁 Body Log"):

    if not body_df.empty:

        st.dataframe(
            body_df,
            use_container_width=True
        )
